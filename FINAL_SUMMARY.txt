   $data = [
       'name' => 'Ma recherche',

   ];

   // Validation avec nouvelle signature
   $result = $search->check($data, [], []);

   if ($result === true) {
       // Validation OK
       $search->store();

       foreach ($errors as $error) {
           echo $error;
       }
   }
