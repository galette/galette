🔄 MODIFICATIONS PRINCIPALES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Héritage d'AbstractEntity
     class SavedSearch extends AbstractEntity { ... }

  2. Mapping avec Attributes
     #[Column(name: 'search_id', insertable: false, updatable: false)]
     protected int $id;

     #[Column(name: 'name')]
     private string $name;

     #[Column(name: 'parameters')]
     private array $parameters = [];

     #[Column(name: 'id_adh')]
     private ?int $author_id = null;

     #[Column(name: 'creation_date')]
     private ?string $creation_date;

     #[Column(name: 'form')]
     private string $form;

  3. Signature check() Standardisée
     AVANT: public function check(array $values): bool
     APRÈS: public function check(array $values, array $required = [], array $disabled = []): bool|array

     → Utilise sanitizeValues()
     → Utilise getLogin() au lieu de $this->login
     → Retourne true ou array<string>

  4. Méthodes Adaptées
     • load() : maintenant publique, retourne static
     • loadFromRS() : maintenant publique, retourne static
     • store() : gère insert ET update
     • remove() : délègue à delete()
     • getErrors() : supprimée (héritée)
